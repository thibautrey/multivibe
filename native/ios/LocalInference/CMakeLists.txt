cmake_minimum_required(VERSION 3.24)
project(MultiVibeLocalInference LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SOURCE} llama)
add_library(MultiVibeLocalInference STATIC MVLlama.mm)
target_compile_options(MultiVibeLocalInference PRIVATE -fobjc-arc)
target_include_directories(MultiVibeLocalInference PRIVATE ${LLAMA_SOURCE}/common ${LLAMA_SOURCE}/include ${LLAMA_SOURCE}/ggml/include ${LLAMA_SOURCE}/vendor)
target_link_libraries(MultiVibeLocalInference PRIVATE llama-common llama)
